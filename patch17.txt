def rep(old, new, must=False):
    global t
    n = t.count(old)
    if n != 1:
        if must:
            sys.exit('PATCH FAILED (%d): %s' % (n, old[:110]))
        print('WARN (%d): %s' % (n, old[:90]))
        return
    t = t.replace(old, new)

    private fun mobSportHook(s: SportInfo?) {
        if (!mobOn || !::mobPD.isInitialized) return
        if (s == null) { if (mobMomPlay) return; mobSportOff(false); return }
        if (mobMomPlay) {
            if (mobSport?.contentId == s.contentId) { mobMomPlay = false; mobSportHeadUpdate() }
            return
        }
        if (!(s.markers || s.matchCode.isNotEmpty())) { mobSportOff(false); return }
        val same = mobSport?.contentId == s.contentId && mobPanelUi != null
        mobSport = s
        if (!same) {
            mobMom.clear(); mobMomState = 0; scoreJson = null; scoreInn = -1
            mobSportTab = if (s.markers) 0 else 1
            mobSportInstall()
        }
        mobSportHeadUpdate()
        if (s.markers && mobMomState == 0) mobMomLoad(s)
        ui.removeCallbacks(mobScoreRun); ui.postDelayed(mobScoreRun, 15000)
    }

    private fun mobScoreLoad() {
        val s = mobSport ?: return
        if (s.matchCode.isEmpty() || mobScoreBusy) return
        mobScoreBusy = true
        Thread {
            val j = Api.get(SCORE_URL + s.matchCode + ".json")
            ui.post {
                mobScoreBusy = false
                if (mobSport?.matchCode != s.matchCode || !inPlayer) return@post
                if (j == null) {
                    if (scoreJson == null) { scoreCol.removeAllViews(); scoreCol.addView(scoreText("Scorecard not available right now", 16f, "#B4B8C8", false)) }
                    return@post
                }
                try {
                    val sy = scoreScroll.scrollY
                    scoreJson = j
                    renderScore2()
                    scoreCol.getChildAt(0)?.visibility = View.GONE
                    scoreScroll.post { scoreScroll.scrollTo(0, sy) }
                } catch (e: Exception) { }
            }
        }.start()
    }

    private fun mobMomLoad(si: SportInfo) {
        if (mobMom.isEmpty()) { mobMomState = 1; mobMomNotify() }
        mobMomAt = SystemClock.uptimeMillis()
        val cid = si.contentId
        Thread {
            val j = Api.get(MOMENTS_URL + cid)
            val arr = j?.optJSONObject("resultObj")?.optJSONArray("containers")
            val list = ArrayList<Ep>()
            val marks = HashMap<String, String>()
            if (arr != null) for (k in 0 until arr.length()) {
                val o = arr.optJSONObject(k) ?: continue
                val ep = parseEp(o) ?: continue
                if (ep.id.isBlank()) continue
                list.add(ep)
                o.optJSONObject("metadata")?.optJSONObject("emfAttributes")?.str("tl_marker")?.let { marks[ep.id] = it.substringAfterLast('|') }
            }
            list.sortByDescending { timeSec(marks[it.id] ?: "") }
            ui.post {
                if (mobSport?.contentId != cid) return@post
                momentMarks.putAll(marks)
                mobMom.clear(); mobMom.addAll(list)
                mobMomState = if (j == null) 3 else 2
                mobMomNotify()
            }
        }.start()
    }
