def rep(old, new, must=True):
    global t
    n = t.count(old)
    if n != 1:
        if must:
            sys.exit('PATCH FAILED (%d): %s' % (n, old[:100]))
        print('WARN (%d): %s' % (n, old[:80]))
        return
    t = t.replace(old, new)

    private fun mobSyncModern(p: ExoPlayer?, show: Boolean) {
        if (p == null || !::mobCtr.isInitialized) return
        val pv = if (pSpin.visibility == View.VISIBLE) View.INVISIBLE else View.VISIBLE
        if (mobPlayBtn.visibility != pv) mobPlayBtn.visibility = pv
        val res = if (p.playbackState == Player.STATE_ENDED || !p.playWhenReady) R.drawable.ic_m_play else R.drawable.ic_m_pause
        if (res != mobPlayRes) {
            val first = mobPlayRes == 0
            mobPlayRes = res
            if (first) mobPlayBtn.setImageResource(res)
            else {
                mobPlayBtn.animate().cancel()
                mobPlayBtn.animate().scaleX(0.7f).scaleY(0.7f).alpha(0.4f).setDuration(80).withEndAction {
                    mobPlayBtn.setImageResource(res)
                    mobPlayBtn.animate().scaleX(1f).scaleY(1f).alpha(1f).setDuration(240)
                        .setInterpolator(android.view.animation.OvershootInterpolator(2f)).start()
                }.start()
            }
        }
        val sk = p.isCurrentMediaItemSeekable
        val sv = if (sk) View.VISIBLE else View.GONE
        if (mobB10.visibility != sv) { mobB10.visibility = sv; mobF10.visibility = sv }
        if (ctrlVisible != mobCtrWas) {
            mobCtrWas = ctrlVisible
            if (ctrlVisible) {
                mobCtr.scaleX = 0.85f; mobCtr.scaleY = 0.85f
                mobCtr.animate().scaleX(1f).scaleY(1f).setDuration(240).setInterpolator(DecelerateInterpolator()).start()
            }
        }
        var n = 0
        for ((b, pl) in mobToolPairs) {
            var vis = pl.visibility == View.VISIBLE
            if (b === mobListBtn && !mobFs) vis = false
            val want = if (vis) View.VISIBLE else View.GONE
            if (b.visibility != want) b.visibility = want
            if (vis) n++
        }
        if (n != mobToolN) {
            mobToolN = n
            (pTitle.parent as? LinearLayout)?.let { tc ->
                (tc.layoutParams as? LinearLayout.LayoutParams)?.let { l -> l.rightMargin = n * dp(46) + dp(14); tc.layoutParams = l }
            }
        }
    }

    private fun mobPdUpdate() {
        if (!::mobPD.isInitialized) return
        val ep = curPlayEp ?: return
        if (lvFull) {
            val cat = liveStartS > 0L
            pdTitle.text = if (cat) ep.title else showTitle
            pdSub.text = (lvCur?.name ?: "") + (if (cat) " \u2022 Catchup" else " \u2022 LIVE")
            pdMeta.text = if (cat) fmtTimeIst(ep.air) + " - " + fmtTimeIst(ep.air + ep.dur * 1000L) else ""
        } else {
            pdTitle.text = showTitle
            val pre = ((if (curPlaySeason > 0) "S$curPlaySeason " else "") + (if (ep.num > 0) "E${ep.num}" else "")).trim()
            pdSub.text = listOf(pre, ep.title).filter { it.isNotBlank() }.joinToString(" \u2022 ")
            val bits = ArrayList<String>()
            fmtDate(ep.air, true).takeIf { it.isNotEmpty() }?.let { bits.add(it) }
            fmtDur(ep.dur).takeIf { it.isNotEmpty() }?.let { bits.add(it) }
            pdMeta.text = bits.joinToString(" \u2022 ")
        }
        pdSub.visibility = if (pdSub.text.isNullOrBlank()) View.GONE else View.VISIBLE
        pdMeta.visibility = if (pdMeta.text.isNullOrBlank()) View.GONE else View.VISIBLE
        val d = if (ep.desc.isNotBlank()) ep.desc else (detailItem?.desc ?: "")
        pdDesc.text = d
        pdDesc.visibility = if (d.isBlank()) View.GONE else View.VISIBLE
        mobPdListUi()
        mobPdNotify()
    }

            h.t3.text = ep.desc
            h.row.setOnClickListener { playEpisode(ep) }
        }
    }
