        fun refresh() {
            pending = false
            items.clear()
            if (td.rows.isEmpty()) items.add(if (td.loading || !td.loaded) "skel" else "err")
            else {
                val hero = td.rows.firstOrNull { it.spotlight && it.items.isNotEmpty() }
                if (hero != null) items.add(hero) else items.add("pad")
                for (r in td.rows) if (!r.spotlight && r.items.isNotEmpty()) items.add(r)
            }
            ad.notifyDataSetChanged()
        }

        fun scheduleRefresh() {
            if (pending) return
            pending = true
            ui.postDelayed({ refresh() }, 160)
        }

        fun ensureLoaded() {
            if (!td.loaded) startLoad(td)
            refresh()
        }

        fun startHero() {
            if (heroOn) return
            heroOn = true
            val r = object : Runnable {
                override fun run() {
                    val h = heroRv
                    if (h == null || !rv.isAttachedToWindow) { heroOn = false; return }
                    if (h.isShown && mobStack.isEmpty()) {
                        val n = h.adapter?.itemCount ?: 0
                        val lm = h.layoutManager as? LinearLayoutManager
                        if (n > 1 && lm != null) {
                            val cur = lm.findFirstVisibleItemPosition()
                            if (cur >= 0) h.smoothScrollToPosition((cur + 1) % n)
                        }
                    }
                    ui.postDelayed(this, 5500)
                }
            }
            ui.postDelayed(r, 5500)
        }

    private fun mobOpenSearch() {
        val list = ArrayList<Item>()
        val trend = ArrayList<Item>()
        val ad = MobGridAd(list)
        val body = LinearLayout(this)
        body.orientation = LinearLayout.VERTICAL
        val et = EditText(this)
        et.hint = "Search shows & movies"; et.setHintTextColor(Color.parseColor("#7A8099")); et.setTextColor(Color.WHITE)
        et.setTextSize(TypedValue.COMPLEX_UNIT_SP, 16f); et.isSingleLine = true
        et.imeOptions = android.view.inputmethod.EditorInfo.IME_ACTION_SEARCH
        et.setPadding(dp(18), dp(10), dp(18), dp(10))
        et.background = GradientDrawable().apply { cornerRadius = dp(26).toFloat(); setColor(Color.argb(255, 28, 30, 42)) }
        val head = mtx("Trending in India", 16f, Color.WHITE, true)
        head.setPadding(dp(16), dp(14), dp(16), dp(6))
        val rv = RecyclerView(this)
        rv.layoutManager = androidx.recyclerview.widget.GridLayoutManager(this, 3)
        rv.adapter = ad
        rv.clipToPadding = false; rv.setPadding(dp(6), dp(4), dp(6), dp(20))
        body.addView(et, LinearLayout.LayoutParams(-1, -2).apply { setMargins(dp(14), dp(4), dp(14), 0) })
        body.addView(head, LinearLayout.LayoutParams(-1, -2))
        body.addView(rv, LinearLayout.LayoutParams(-1, 0, 1f))
        val tok = IntArray(1)
        val runH = arrayOfNulls<Runnable>(1)
        fun show(l: List<Item>, h: String) {
            list.clear(); list.addAll(l); ad.notifyDataSetChanged(); head.text = h
        }
        fun loadTrend() {
            Thread {
                val j = Api.get(TRENDING_URL)
                val l = ArrayList<Item>()
                collectSearch(j?.optJSONObject("resultObj")?.optJSONArray("containers"), l, HashSet(), 0)
                ui.post {
                    trend.clear(); trend.addAll(l)
                    if (et.isAttachedToWindow && et.text.toString().trim().length < 2) show(l, "Trending in India")
                }
            }.start()
        }
        et.addTextChangedListener(object : android.text.TextWatcher {
            override fun beforeTextChanged(s: CharSequence?, a: Int, b: Int, c: Int) { }
            override fun onTextChanged(s: CharSequence?, a: Int, b: Int, c: Int) { }
            override fun afterTextChanged(s: android.text.Editable?) {
                tok[0]++
                val my = tok[0]
                val q = (s?.toString() ?: "").trim()
                runH[0]?.let { ui.removeCallbacks(it) }
                if (q.length < 2) { if (trend.isNotEmpty()) show(trend, "Trending in India") else loadTrend(); return }
                val r = Runnable {
                    Thread {
                        val j = Api.get(SEARCH_URL + java.net.URLEncoder.encode(q, "UTF-8") + "&from=0&to=30")
                        val l = ArrayList<Item>()
                        collectSearch(j?.optJSONObject("resultObj")?.optJSONArray("containers"), l, HashSet(), 0)
                        ui.post {
                            if (my == tok[0] && et.isAttachedToWindow) {
                                if (j == null) show(emptyList(), "Something went wrong. Check your internet connection.")
                                else show(l, if (l.isEmpty()) "No results found" else "Results")
                            }
                        }
                    }.start()
                }
                runH[0] = r
                ui.postDelayed(r, 400)
            }
        })
        et.setOnEditorActionListener { _, _, _ -> mobHideKb(); true }
        mobPush(mobShell("Search", body))
        loadTrend()
        et.requestFocus()
        ui.postDelayed({
            try {
                val imm = getSystemService(INPUT_METHOD_SERVICE) as android.view.inputmethod.InputMethodManager
                imm.showSoftInput(et, 0)
            } catch (e: Exception) { }
        }, 300)
    }

        if ((item.sub != SUB_SHOW && item.sub != SUB_MOVIE_BUNDLE) || item.cid.isBlank()) return
        detailMovie = item.sub == SUB_MOVIE_BUNDLE
        detailItem = item
        movieEp = null; trailerEp = null; trailerPlayUrl = null; lastBtn = null
        epToken++
        seasons.clear(); eps.clear(); epAdapter.notifyDataSetChanged()
        curSeason = -1; epFrom = 0; epHasMore = true; epLoading = false; latestEp = null
        homeShowId = item.cid
        val d = MobDetail(item)
        mobDetail = d
        d.root.tag = d
        mobPush(d.root)
        d.preview(); d.load()
    }

        fun load() {
            Thread {

                ui.post { if (mobDetail === this) bind(j) }

        fun bind(j: JSONObject?) {
            val show = j?.optJSONObject("resultObj")?.optJSONArray("containers")?.optJSONObject(0)
            if (show == null) { descTv.text = "Could not load details. Check your internet connection."; return }
            val m = show.optJSONObject("metadata") ?: JSONObject()
            val e = m.optJSONObject("emfAttributes") ?: JSONObject()
            val title = m.str("title") ?: item.title
            showTitle = title
            val desc = m.str("shortDescription") ?: m.str("longDescription") ?: item.desc
            val movieChild = if (mv) childBySubtype(show, "MOVIE") else null
            val trailerChild = if (mv) childBySubtype(show, "MOVIE_TRAILER") else null
            val mm = movieChild?.optJSONObject("metadata")
            val me = mm?.optJSONObject("emfAttributes")
            val parts = ArrayList<String>()
            (m.str("languageLabel") ?: e.str("region_of_origin"))?.let { parts.add(it) }
            m.str("year")?.let { parts.add(it) }
            m.str("pcVodLabel")?.let { parts.add(it) }
            if (mv) {
                val d = fmtDur(mm?.optInt("duration") ?: 0)
                if (d.isNotEmpty()) parts.add(d)
            } else {
                val sn = m.optInt("seasonCount")
                if (sn > 1) parts.add("$sn Seasons")
                val ec = m.optInt("episodeCount")
                if (ec > 0) parts.add("$ec Episodes")
            }
            m.optJSONArray("genres")?.let { g ->
                val l = (0 until g.length()).map { k -> g.optString(k) }.filter { s -> s.isNotBlank() }
                if (l.isNotEmpty()) parts.add(l.joinToString(", "))
            }
            val metaS = parts.joinToString(" \u2022 ")
            metaTv.text = metaS; descTv.text = desc; titleTv.text = title
            val tag = e.str("highlight_text") ?: (if (mv) m.optJSONObject("badges")?.str("primary_tag") else null)
            if (tag != null) { tagTv.text = tag; tagTv.visibility = View.VISIBLE }
            detailItem = item.copy(title = title, desc = desc, meta = metaS,
                bg = rawBgOf(show) ?: item.bg, logo = detLogoUrl(show) ?: item.logo,
                portrait = e.str("portrait_thumb") ?: item.portrait, landscape = e.str("thumbnail") ?: item.landscape)
            bgBigReq(detBgUrl(show) ?: img(item.bg ?: item.landscape, W_BG)).into(banner)
            val lg = detLogoUrl(show) ?: mergeEnr(item).logo
            if (lg != null) {
                logo.visibility = View.VISIBLE; titleTv.visibility = View.GONE
                logoReq(lg, (mobW * 0.62f).toInt(), dp(84)).into(logo)
            }
            val tUrl = trailerUrlAny(show)
            if (tUrl != null) {
                val tm = trailerChild?.optJSONObject("metadata")
                val tid = trailerIdFrom(show) ?: (tm?.optString("contentId") ?: "")
                trailerPlayUrl = tUrl
                trailerEp = Ep(tid, "Trailer", 0, desc, e.str("thumbnail") ?: item.landscape, tm?.optInt("duration") ?: 0, 0L)
                trailerBtn.visibility = View.VISIBLE
            } else { trailerPlayUrl = null; trailerEp = null; trailerBtn.visibility = View.GONE }
            if (mv) {
                val mid = mm?.optString("contentId").orEmpty()
                movieEp = if (movieChild != null && mid.isNotBlank())
                    Ep(mid, "", 0, desc, me?.str("thumbnail") ?: e.str("thumbnail") ?: item.landscape, mm?.optInt("duration") ?: 0, mm?.optLong("originalAirDate") ?: 0L)
                else null
                extra.removeAllViews()
                for ((k, v) in movieExtraRows(e, me)) {
                    if (v.isBlank()) continue
                    val sb = SpannableStringBuilder()
                    sb.append(k).append("   ")
                    sb.setSpan(ForegroundColorSpan(Color.parseColor("#9AA0B4")), 0, k.length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
                    sb.setSpan(StyleSpan(Typeface.BOLD), 0, k.length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
                    sb.append(v)
                    val tv = mtx("", 12.5f, Color.parseColor("#E4E4EA"))
                    tv.text = sb; tv.maxLines = if (k == "Cast") 3 else 2; tv.ellipsize = TextUtils.TruncateAt.END
                    extra.addView(tv, LinearLayout.LayoutParams(-1, -2).apply { topMargin = dp(6) })
                }
            } else {
                val data = parseSeasons(show)
                seasons.clear(); seasons.addAll(data.seasons)
                data.latest?.let { latestEp = it }
                buildChips()
                if (seasons.isNotEmpty()) { selectSeason(0, false); styleChips() }
            }
            updateList(); updateAct()
        }
    }

    // ---------------------------------------------------------------- mobile Live TV (channel grid, tap = full screen player)
    private var mobLiveBox: FrameLayout? = null
    private var mobLiveAd: RecyclerView.Adapter<RecyclerView.ViewHolder>? = null
    private val mobLiveItems = ArrayList<Any>()
    private var mobLivePend = false
    private var mobLivePoll = false
    private var mobLiveMsg: TextView? = null

    private fun mobLiveBuild() {
        mobLiveItems.clear()
        for (r in lvRows) { mobLiveItems.add(r.title); mobLiveItems.addAll(r.items) }
        mobLiveAd?.notifyDataSetChanged()
        mobLiveMsg?.visibility = if (mobLiveItems.isEmpty()) View.VISIBLE else View.GONE
    }

    private fun mobLiveEpg() {
        if (mobLiveBox == null) return
        if (mobLiveItems.isEmpty() && lvRows.isNotEmpty()) { mobLiveBuild(); return }
        if (mobLivePend) return
        mobLivePend = true
        ui.postDelayed({ mobLivePend = false; mobLiveAd?.notifyDataSetChanged() }, 400)
    }

    private fun mobLiveTick() {
        if (mobTab != 5 || !mobOn) { mobLivePoll = false; return }
        if (lvLoaded && mobLiveItems.isEmpty()) mobLiveBuild()
        if (lvLoaded) { for (c in lvChans) { val pr = lvProgNow(c); if (pr == null) lvEnsureEpg(c) }; mobLiveAd?.notifyDataSetChanged() }
        ui.postDelayed({ mobLiveTick() }, if (lvLoaded) 30000L else 600L)
    }

    inner class MobLiveVH(val box: LinearLayout, val iv: ImageView, val t1: TextView, val t2: TextView) : RecyclerView.ViewHolder(box)
    inner class MobLiveHead(val tv: TextView) : RecyclerView.ViewHolder(tv)

    private fun mobShowLive() {
        var box = mobLiveBox
        if (box == null) {
            val b = FrameLayout(this)
            b.setBackgroundColor(mBg)
            val rv = RecyclerView(this)
            val gm = androidx.recyclerview.widget.GridLayoutManager(this, 3)
            gm.spanSizeLookup = object : androidx.recyclerview.widget.GridLayoutManager.SpanSizeLookup() {
                override fun getSpanSize(p: Int): Int = if (mobLiveItems.getOrNull(p) is String) 3 else 1
            }
            rv.layoutManager = gm
            rv.clipToPadding = false; rv.setPadding(dp(6), dp(60), dp(6), dp(16))
            rv.itemAnimator = null; rv.overScrollMode = View.OVER_SCROLL_NEVER
            val cw = (mobW - dp(12)) / 3 - dp(10)
            val ch = cw * 3 / 2
            val ad = object : RecyclerView.Adapter<RecyclerView.ViewHolder>() {
                override fun getItemCount() = mobLiveItems.size
                override fun getItemViewType(position: Int) = if (mobLiveItems[position] is String) 0 else 1
                override fun onCreateViewHolder(p: ViewGroup, t: Int): RecyclerView.ViewHolder {
                    val ctx = this@MainActivity
                    if (t == 0) {
                        val tv = mtx("", 17f, Color.WHITE, true)
                        tv.setPadding(dp(10), dp(18), dp(10), dp(8))
                        tv.layoutParams = RecyclerView.LayoutParams(-1, -2)
                        return MobLiveHead(tv)
                    }
                    val bx = LinearLayout(ctx)
                    bx.orientation = LinearLayout.VERTICAL; bx.isClickable = true
                    bx.layoutParams = RecyclerView.LayoutParams(-1, -2)
                    bx.setPadding(dp(5), dp(5), dp(5), dp(5))
                    val iv = ImageView(ctx)
                    iv.setBackgroundColor(Color.rgb(26, 28, 42)); mobRound(iv, dp(8))
                    val t1 = mtx("", 12.5f, Color.WHITE, true); t1.maxLines = 1; t1.ellipsize = TextUtils.TruncateAt.END
                    val t2 = mtx("", 11f, Color.parseColor("#9AA0B4")); t2.maxLines = 1; t2.ellipsize = TextUtils.TruncateAt.END
                    bx.addView(iv, LinearLayout.LayoutParams(-1, ch))
                    bx.addView(t1, LinearLayout.LayoutParams(-1, -2).apply { topMargin = dp(6) })
                    bx.addView(t2, LinearLayout.LayoutParams(-1, -2))
                    return MobLiveVH(bx, iv, t1, t2)
                }
                override fun onBindViewHolder(h: RecyclerView.ViewHolder, pos: Int) {
                    val o = mobLiveItems[pos]
                    if (h is MobLiveHead) { h.tv.text = o as String; return }
                    val v = h as MobLiveVH
                    val c = o as LvCh
                    val pr = lvProgNow(c)
                    val poster = pr?.poster
                    if (poster != null) {
                        v.iv.scaleType = ImageView.ScaleType.CENTER_CROP; v.iv.setPadding(0, 0, 0, 0)
                        Glide.with(this@MainActivity).load(poster).override(cw, ch).centerCrop().format(DecodeFormat.PREFER_RGB_565)
                            .diskCacheStrategy(DiskCacheStrategy.ALL).into(v.iv)
                    } else {
                        v.iv.scaleType = ImageView.ScaleType.FIT_CENTER; v.iv.setPadding(dp(14), dp(14), dp(14), dp(14))
                        if (c.logo.isNotBlank()) Glide.with(this@MainActivity).load(c.logo).diskCacheStrategy(DiskCacheStrategy.ALL).fitCenter().into(v.iv)
                        else v.iv.setImageDrawable(null)
                        lvEnsureEpg(c)
                    }
                    v.t1.text = pr?.name ?: c.name
                    v.t2.text = if (pr != null) fmtTimeIst(pr.start) + " - " + fmtTimeIst(pr.end) else c.name
                    v.box.setOnClickListener { lvPlayFull(c, v.box) }
                }
            }
            @Suppress("UNCHECKED_CAST")
            mobLiveAd = ad as RecyclerView.Adapter<RecyclerView.ViewHolder>
            rv.adapter = ad
            b.addView(rv, lp(-1, -1))
            val msg = mtx("Loading channels...", 15f, Color.parseColor("#B4B8C8"))
            b.addView(msg, lp(-2, -2, Gravity.CENTER))
            mobLiveMsg = msg
            mobPages.addView(b, lp(-1, -1))
            mobLiveBox = b
            box = b
        }
        box.visibility = View.VISIBLE
        if (!lvLoaded) lvLoad() else if (mobLiveItems.isEmpty()) mobLiveBuild()
        if (!mobLivePoll) { mobLivePoll = true; ui.postDelayed({ mobLiveTick() }, 400) }
    }

    private fun mobTap(x: Float, w: Float) {
        val now = SystemClock.uptimeMillis()
        val side = if (x < w * 0.35f) 0 else if (x > w * 0.65f) 2 else 1
        if (side != 1 && gAcc > 0 && now - gLastTap < 700) { mobSeekTap(side == 2); gLastTap = now; return }
        if (now - gLastTap < 300 && gLastSide == side) {
            ui.removeCallbacks(gTapRun); gLastTap = now
            if (side == 1) togglePause() else mobSeekTap(side == 2)
            return
        }
        gLastTap = now; gLastSide = side
        ui.postDelayed(gTapRun, 300)
    }

    }
