class MobSeek(c: Context) : View(c) {
    var dur = 0L
    var pos = 0L
    var buf = 0L
    var dragging = false
    var onStart: (() -> Unit)? = null
    var onScrub: ((Long, Boolean) -> Unit)? = null
    private val pt = Paint(Paint.ANTI_ALIAS_FLAG)
    private var grow = 0f
    private val d = c.resources.displayMetrics.density
    private fun frac(x: Float): Float {
        val l = 14f * d
        val r = width - 14f * d
        return ((x - l) / max(1f, r - l)).coerceIn(0f, 1f)
    }
    private fun setG(to: Float) {
        ValueAnimator.ofFloat(grow, to).apply {
            duration = 140L
            addUpdateListener { grow = it.animatedValue as Float; invalidate() }
            start()
        }
    }
    private fun upd(x: Float, fin: Boolean) {
        pos = (frac(x) * dur).toLong()
        invalidate()
        onScrub?.invoke(pos, fin)
    }
    override fun onDraw(cv: Canvas) {
        val l = 14f * d
        val r = width - 14f * d
        val cy = height / 2f
        val th = (3f + 2f * grow) * d
        pt.style = Paint.Style.FILL
        pt.color = Color.argb(70, 255, 255, 255)
        cv.drawRoundRect(l, cy - th / 2, r, cy + th / 2, th, th, pt)
        if (dur > 0L) {
            val bx = l + (r - l) * (buf.toFloat() / dur).coerceIn(0f, 1f)
            pt.color = Color.argb(110, 255, 255, 255)
            cv.drawRoundRect(l, cy - th / 2, bx, cy + th / 2, th, th, pt)
            val px = l + (r - l) * (pos.toFloat() / dur).coerceIn(0f, 1f)
            pt.color = Color.parseColor("#FFC107")
            cv.drawRoundRect(l, cy - th / 2, px, cy + th / 2, th, th, pt)
            pt.color = Color.WHITE
            cv.drawCircle(px, cy, (6f + 4f * grow) * d, pt)
        }
    }
    override fun onTouchEvent(e: MotionEvent): Boolean {
        if (dur <= 0L) return false
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> { parent?.requestDisallowInterceptTouchEvent(true); dragging = true; setG(1f); onStart?.invoke(); upd(e.x, false) }
            MotionEvent.ACTION_MOVE -> if (dragging) upd(e.x, false)
            MotionEvent.ACTION_UP -> if (dragging) { upd(e.x, true); dragging = false; setG(0f) }
            MotionEvent.ACTION_CANCEL -> { dragging = false; setG(0f) }
        }
        return true
    }
}

    // gestures: tap = controls, double tap = pause / +-10s, hold = 2x, swipe = seek, left = brightness, right = volume
    private fun mobGestTouch(ev: MotionEvent): Boolean {
        val a = ev.actionMasked
        val w = max(1, playerLayer.width).toFloat()
        val h = max(1, playerLayer.height).toFloat()
        if (a == MotionEvent.ACTION_DOWN) {
            gMode = 0; gX0 = ev.rawX; gY0 = ev.rawY; gLX = gX0; gLY = gY0
            ui.postDelayed(gLongRun, 450)
            return true
        }
        if (a == MotionEvent.ACTION_MOVE) {
            val dx = ev.rawX - gX0
            val dy = ev.rawY - gY0
            val p = mainPlayer
            if (gMode == 0 && (abs(dx) > dp(14) || abs(dy) > dp(14))) {
                ui.removeCallbacks(gLongRun); ui.removeCallbacks(gTapRun)
                if (abs(dx) > abs(dy)) {
                    if (p != null && p.isCurrentMediaItemSeekable) { gMode = 1; gBase = p.currentPosition } else gMode = 6
                } else if (ev.x < w / 2f) gMode = 2
                else {
                    gMode = 3
                    val am = getSystemService(AUDIO_SERVICE) as android.media.AudioManager
                    gVolF = am.getStreamVolume(android.media.AudioManager.STREAM_MUSIC).toFloat()
                }
            }
            if (gMode == 1 && p != null) {
                val mx = if (p.duration > 0) p.duration - 1000L else Long.MAX_VALUE
                gSeekTo = (gBase + (dx / w * 120000f).toLong()).coerceIn(0L, mx)
                val dl = (gSeekTo - gBase) / 1000L
                mobHud(R.drawable.ic_m_play, (if (dl >= 0) "+" else "") + dl + "s  \u2022  " + mobFmt(gSeekTo))
            } else if (gMode == 2) {
                val lpw = window.attributes
                var cur = lpw.screenBrightness
                if (cur < 0f) cur = try { android.provider.Settings.System.getInt(contentResolver, android.provider.Settings.System.SCREEN_BRIGHTNESS) / 255f } catch (e: Exception) { 0.5f }
                cur = (cur + (gLY - ev.rawY) / (h * 0.7f)).coerceIn(0.02f, 1f)
                lpw.screenBrightness = cur; window.attributes = lpw
                mobHud(R.drawable.ic_brightness, ((cur * 100).toInt()).toString() + "%")
            } else if (gMode == 3) {
                val am = getSystemService(AUDIO_SERVICE) as android.media.AudioManager
                val mx = am.getStreamMaxVolume(android.media.AudioManager.STREAM_MUSIC)
                gVolF = (gVolF + (gLY - ev.rawY) / (h * 0.7f) * mx).coerceIn(0f, mx.toFloat())
                am.setStreamVolume(android.media.AudioManager.STREAM_MUSIC, Math.round(gVolF), 0)
                mobHud(R.drawable.ic_m_audio, ((gVolF / mx * 100).toInt()).toString() + "%")
            }
            gLX = ev.rawX; gLY = ev.rawY
            return true
        }
        if (a == MotionEvent.ACTION_UP || a == MotionEvent.ACTION_CANCEL) {
            ui.removeCallbacks(gLongRun)
            when (gMode) {
                1 -> mainPlayer?.seekTo(gSeekTo)
                4 -> mobHoldSpeed(false)
                0 -> if (a == MotionEvent.ACTION_UP) mobTap(ev.x, w)
            }
            gMode = 0
        }
        return true
    }

    private fun mobAudioRows(p: ExoPlayer, done: () -> Unit): List<MobRowD> {
        val out = ArrayList<MobRowD>()
        if (audioOpts.isNotEmpty()) {
            for (ao in audioOpts) out.add(MobRowD(ao.label, "", 0, ao.code.equals(curAudio, true), false) { switchAudio(ao.code); done() })
            return out
        }
        class AC(val g: Tracks.Group, val i: Int, val br: Int, var sel: Boolean)
        val m = LinkedHashMap<String, AC>()
        var n = 0
        for (g in p.currentTracks.groups) if (g.type == C.TRACK_TYPE_AUDIO) for (i in 0 until g.length) {
            if (!g.isTrackSupported(i)) continue
            val f = g.getTrackFormat(i)
            val lb = trackLabel(f, n++)
            val sel = g.isTrackSelected(i)
            val cur = m[lb]
            if (cur == null) m[lb] = AC(g, i, f.bitrate, sel)
            else if (f.bitrate > cur.br) m[lb] = AC(g, i, f.bitrate, cur.sel || sel)
            else if (sel) cur.sel = true
        }
        for ((lb, c) in m) out.add(MobRowD(lb, "", 0, c.sel, false) { selectAudio(c.g, c.i); rememberTrack(c.g.getTrackFormat(c.i)); done() })
        return out
    }

    private fun mobSubRows(p: ExoPlayer, done: () -> Unit): List<MobRowD> {
        val out = ArrayList<MobRowD>()
        val off = p.trackSelectionParameters.disabledTrackTypes.contains(C.TRACK_TYPE_TEXT)
        val tl = ArrayList<Triple<Tracks.Group, Int, String>>()
        for (g in p.currentTracks.groups) if (g.type == C.TRACK_TYPE_TEXT) for (i in 0 until g.length) {
            if (g.isTrackSupported(i)) tl.add(Triple(g, i, trackLabel(g.getTrackFormat(i), tl.size)))
        }
        if (tl.isEmpty()) return out
        val anySel = !off && tl.any { it.first.isTrackSelected(it.second) }
        out.add(MobRowD("Off", "", 0, !anySel, false) { selectText(null, 0); done() })
        for (x in tl) out.add(MobRowD(x.third, "", 0, !off && x.first.isTrackSelected(x.second), false) { selectText(x.first, x.second); done() })
        return out
    }
