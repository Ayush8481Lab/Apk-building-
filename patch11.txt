def rep(old, new):

    n = t.count(old)
    if n != 1:

    t = t.replace(old, new)

fun lvHexToB64(h: String): String {
    val s = h.trim()
    val b = ByteArray(s.length / 2) { s.substring(it * 2, it * 2 + 2).toInt(16).toByte() }
    return android.util.Base64.encodeToString(b, android.util.Base64.URL_SAFE or android.util.Base64.NO_PADDING or android.util.Base64.NO_WRAP)
}

// clearkey response: "license" json as it is, or built from keyId + key (hex)
fun lvClearKeyJson(c: LvCh): String? {
    if (c.license.startsWith("{")) return c.license
    if (c.keyId.isNotBlank() && c.key.isNotBlank()) {
        return try {
            JSONObject().put("keys", JSONArray().put(JSONObject()
                .put("kty", "oct").put("k", lvHexToB64(c.key)).put("kid", lvHexToB64(c.keyId))))
                .put("type", "temporary").toString()
        } catch (e: Exception) { null }
    }
    return null
}

fun lvBuildSource(ch: LvCh): MediaSource {
    LvCookies.init()
    val dsf = DefaultHttpDataSource.Factory()
        .setUserAgent(LV_UA)
        .setAllowCrossProtocolRedirects(true)
        .setConnectTimeoutMs(15000)
        .setReadTimeoutMs(20000)
    val mb = MediaItem.Builder().setUri(ch.url).setMimeType(MimeTypes.APPLICATION_MPD)
    if (ch.licUrl.isNotBlank()) {
        mb.setDrmConfiguration(MediaItem.DrmConfiguration.Builder(C.WIDEVINE_UUID).setLicenseUri(ch.licUrl).build())
        return androidx.media3.exoplayer.dash.DashMediaSource.Factory(dsf).createMediaSource(mb.build())
    }
    val kj = lvClearKeyJson(ch)
    if (kj != null) {
        val mgr = androidx.media3.exoplayer.drm.DefaultDrmSessionManager.Builder()
            .setUuidAndExoMediaDrmProvider(C.CLEARKEY_UUID, androidx.media3.exoplayer.drm.FrameworkMediaDrm.DEFAULT_PROVIDER)
            .build(androidx.media3.exoplayer.drm.LocalMediaDrmCallback(kj.toByteArray()))
        return androidx.media3.exoplayer.dash.DashMediaSource.Factory(dsf)
            .setDrmSessionManagerProvider { mgr }
            .createMediaSource(mb.build())
    }
    return androidx.media3.exoplayer.dash.DashMediaSource.Factory(dsf).createMediaSource(mb.build())
}

fun lvHttp(url: String): JSONObject? = try {
    val req = Request.Builder().url(url).header("User-Agent", STREAM_UA).header("Accept", "application/json").build()
    Net.client.newCall(req).execute().use { r -> if (r.isSuccessful) JSONObject(r.body!!.string()) else null }
} catch (e: Exception) { null }

fun lvParse(j: JSONObject): Pair<Long, List<LvCh>> {
    val out = ArrayList<LvCh>()
    val arr = j.optJSONArray("channels")
    if (arr != null) for (i in 0 until arr.length()) {
        val o = arr.optJSONObject(i) ?: continue
        val url = o.str("url") ?: continue
        val id = o.str("id") ?: continue
        var lic = (o.str("license") ?: "").trim()
        var lu = o.str("licenseurl") ?: o.str("licenseUrl") ?: ""
        if (lic.startsWith("http")) { lu = lic; lic = "" }
        out.add(LvCh(id, o.str("name") ?: id, (o.str("category") ?: "Others").trim().ifEmpty { "Others" }, url,
            o.str("keyId") ?: "", o.str("key") ?: "", lic, lu, o.str("logo") ?: ""))
    }
    return Pair(j.optLong("expires_at", 0L) * 1000L, out)
}

fun lvImg(p: String?): String? {
    if (p.isNullOrBlank()) return null
    return if (p.startsWith("http")) p else LV_IMG + "/" + p.trimStart('/')
}

fun lvFetchProgs(id: String, off: Int): List<LvProg>? {
    return try {
        val req = Request.Builder().url(EPG_URL + id + "&offset=" + off)
            .header("User-Agent", STREAM_UA).header("Accept", "application/json").build()
        Net.client.newCall(req).execute().use { r ->
            if (!r.isSuccessful) null
            else {
                val arr = JSONObject(r.body!!.string()).optJSONArray("epg")
                val out = ArrayList<LvProg>()
                if (arr != null) for (i in 0 until arr.length()) {
                    val o = arr.optJSONObject(i) ?: continue
                    val g = o.optJSONArray("showGenre")
                    val genre = if (g == null) "" else (0 until g.length()).map { k -> g.optString(k) }.filter { it.isNotBlank() }.joinToString(", ")
                    val assets = o.optJSONObject("assets")
                    fun asset(k: String): String? = assets?.optJSONObject(k)?.str("program") ?: assets?.optJSONObject(k)?.str("episode")
                    out.add(LvProg(o.optString("showname"), o.str("description") ?: o.optString("episode_desc"), genre, o.optString("showCategory"),
                        lvImg(asset("16:9") ?: o.str("episodeThumbnail")), lvImg(asset("2:3") ?: o.str("episodePoster")),
                        o.optLong("startEpoch"), o.optLong("endEpoch")))
                }
                out.sortedBy { it.start }
            }
        }
    } catch (e: Exception) { null }
}

// the day list that contains "now" (tries the likely offset first, then the neighbours)
fun lvFetchCurrent(id: String, base: Int): List<LvProg>? {
    val now = System.currentTimeMillis()
    var first: List<LvProg>? = null
    for (o in listOf(base, 0, 1, -1).distinct()) {
        val l = lvFetchProgs(id, o) ?: continue
        if (first == null) first = l
        if (l.any { it.start <= now && now < it.end }) return l
    }
    return first
}

    private fun lvLoad() {
        if (lvLoading) return
        lvLoading = true
        if (lvRows.isEmpty()) { lvStatus.text = "Loading channels..."; lvStatus.visibility = View.VISIBLE }
        Thread {
            val j = lvHttp(LV_URL)

            ui.post {

                if (parsed == null || parsed.second.isEmpty()) {
                    if (lvRows.isEmpty()) { lvStatus.text = "Could not load channels\nSelect Live TV again to retry"; lvStatus.visibility = View.VISIBLE }
                    lvScheduleRefresh(60000L)
                    return@post
                }
                lvApply(parsed.first, parsed.second)
            }
        }.start()
    }

        ui.postDelayed(r, delay)
    }

    private fun lvApply(exp: Long, list: List<LvCh>) {
        lvExpMs = exp
        val same = lvLoaded && lvChans.map { it.id }.toSet() == list.map { it.id }.toSet()
        if (same) {
            for (n in list) lvChans.firstOrNull { it.id == n.id }?.let { o ->
                o.name = n.name; o.url = n.url; o.keyId = n.keyId; o.key = n.key
                o.license = n.license; o.licUrl = n.licUrl; o.logo = n.logo
            }
        } else {
            lvChans = list
            val m = LinkedHashMap<String, ArrayList<LvCh>>()
            val titles = HashMap<String, String>()
            for (c in list) {
                val k = c.cat.lowercase()
                m.getOrPut(k) { ArrayList() }.add(c)
                if (!titles.containsKey(k)) titles[k] = c.cat
            }
            lvRows.clear()
            for ((k, v) in m) lvRows.add(LvRow(titles[k] ?: k, v))
            lvRv.adapter?.notifyDataSetChanged()
            lvSnap = -1
            if (lvActive()) lvFocusFirst(12)
        }
        lvLoaded = true
        lvStatus.visibility = View.GONE
        val d = if (exp > 0L) exp - 600000L - System.currentTimeMillis() else 1800000L
        lvScheduleRefresh(d.coerceIn(60000L, 6 * 3600000L))
    }

    // up / down: first card of the previous / next category row
    private fun lvVertical(dir: Int): Boolean {

    private fun lvFocusRowFirst(target: Int, tries: Int) {
        val rvh = lvRv.findViewHolderForAdapterPosition(target) as? LvRowVH
        if (rvh == null) {
            if (tries == 14) lvSmooth(target)
            if (tries > 0) ui.postDelayed({ lvFocusRowFirst(target, tries - 1) }, 60)
            return
        }
        val first = rvh.rv.findViewHolderForAdapterPosition(0)?.itemView
        if (first != null) first.requestFocus()
        else {
            rvh.rv.scrollToPosition(0)
            if (tries > 0) ui.postDelayed({ lvFocusRowFirst(target, tries - 1) }, 40)
        }
    }
