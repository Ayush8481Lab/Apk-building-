import sys, pathlib
P = pathlib.Path('app/src/main/java/com/example/livtv/MainActivity.kt')
t = P.read_text(encoding='utf-8')

def rep(old, new, must=False):
    global t
    n = t.count(old)
    if n != 1:
        if must:
            sys.exit('PATCH FAILED (%d): %s' % (n, old[:110]))
        print('WARN (%d): %s' % (n, old[:90]))
        return
    t = t.replace(old, new)

def repall(old, new):
    global t
    n = t.count(old)
    if n == 0:
        print('WARN (0): %s' % old[:90]); return
    t = t.replace(old, new)
