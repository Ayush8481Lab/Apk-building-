import sys, pathlib
P = pathlib.Path('app/src/main/java/com/example/livtv/MainActivity.kt')
t = P.read_text(encoding='utf-8')

def hook(old, new, required=True):
    global t
    if old not in t:
        if required:
            sys.exit('PATCH FAILED (0 matches): ' + old[:100])
        print('WARN: ' + old[:80])
        return
    t = t.replace(old, new, 1)

# ---- start the gate after the UI is built
hook('setContentView(root)', 'setContentView(root)\n        ui.post { gateStart() }')

# ---- while the gate is up: normal touch + keys (no swipe->dpad), back = exit, no trailers
hook('override fun dispatchKeyEvent(ev: KeyEvent): Boolean {',
     'override fun dispatchKeyEvent(ev: KeyEvent): Boolean {\n        if (gateOn) return super.dispatchKeyEvent(ev)')
hook('override fun dispatchTouchEvent(ev: MotionEvent): Boolean {',
     'override fun dispatchTouchEvent(ev: MotionEvent): Boolean {\n        if (gateOn) return super.dispatchTouchEvent(ev)')
hook('override fun onBackPressed() {',
     'override fun onBackPressed() {\n        if (gateOn) { finish(); return }')
hook('private fun playTrailer(url: String, spotLoop: Boolean) {',
     'private fun playTrailer(url: String, spotLoop: Boolean) {\n        if (gateOn) return', False)

GATE = r'''// ---------------------------------------------------------------- Update 20 : access token gate
    private var gateOn = false
    private var gateView: FrameLayout? = null
    private var gateBusy = false

    private fun gateTv(): Boolean =
        (getSystemService(UI_MODE_SERVICE) as UiModeManager).currentModeType == Configuration.UI_MODE_TYPE_TELEVISION

    // "TVAPP - <model no> <name>"  /  "Mobileapp - <model no> <name>"  (header only used for the token check)
    private fun gateUa(): String {
        val s = (if (gateTv()) "TVAPP - " else "Mobileapp - ") + android.os.Build.MODEL + " " + android.os.Build.MANUFACTURER
        return s.replace(Regex("[^\\x20-\\x7E]"), "").trim()
    }

    // true = valid, false = invalid, null = could not check (no internet / server error)
    private fun gateCheck(token: String): Boolean? {
        return try {
            val req = Request.Builder().url("https://acessviayush.pages.dev/dev/" + Uri.encode(token))
                .header("User-Agent", gateUa()).build()
            Net.client.newCall(req).execute().use { r ->
                val b = r.body?.string().orEmpty().trim().trim('"').lowercase()
                if (r.isSuccessful) b == "true" else if (r.code < 500) false else null
            }
        } catch (e: Exception) { null }
    }

    private fun gateStart() {
        val tk = getSharedPreferences("liv", Context.MODE_PRIVATE).getString("acc_token", "") ?: ""
        if (tk.isBlank()) { gateShow(); return }
        // saved token: app is already open, verify in the background
        Thread {
            val ok = gateCheck(tk)
            if (ok == false) ui.post { gateLock() }
        }.start()
    }

    // token no longer valid: close everything, forget the token, show the locked page
    private fun gateLock() {
        getSharedPreferences("liv", Context.MODE_PRIVATE).edit().remove("acc_token").apply()
        gateOn = true
        heroRun?.let { ui.removeCallbacks(it) }
        heroToken++
        stopTrailer(false)
        try { mainPlayer?.let { it.stop(); it.release() } } catch (e: Exception) { }
        mainPlayer = null
        Prefetch.paused = true
        gateShow()
    }

    private fun gateShow() {
        gateOn = true
        if (splashOn) endSplash()
        stopTrailer(false)
        gateView?.let { root.removeView(it) }
        window.setSoftInputMode(WindowManager.LayoutParams.SOFT_INPUT_ADJUST_PAN)
        val f = FrameLayout(this).apply { setBackgroundColor(Color.argb(255, 5, 6, 12)); isClickable = true }
        val col = LinearLayout(this).apply { orientation = LinearLayout.VERTICAL; gravity = Gravity.CENTER_HORIZONTAL }
        val lock = TextView(this).apply { text = "\uD83D\uDD12"; gravity = Gravity.CENTER }
        tx(lock, 40f)
        val t1 = TextView(this).apply { text = "Unauthorized access"; setTextColor(Color.parseColor("#FF5252")); typeface = Typeface.DEFAULT_BOLD }
        tx(t1, 28f)
        val t2 = TextView(this).apply { text = "Enter a valid access token to continue"; setTextColor(Color.parseColor("#B4B8C8")); gravity = Gravity.CENTER }
        tx(t2, 16f)
        val et = EditText(this).apply {
            hint = "Access token"
            setHintTextColor(Color.parseColor("#7A8099")); setTextColor(Color.WHITE)
            setSingleLine(true)
            inputType = android.text.InputType.TYPE_CLASS_TEXT or android.text.InputType.TYPE_TEXT_FLAG_NO_SUGGESTIONS
            imeOptions = android.view.inputmethod.EditorInfo.IME_ACTION_DONE
            setPadding(u(18), u(12), u(18), u(12))
            gravity = Gravity.CENTER_VERTICAL
        }
        tx(et, 18f)
        fun stEt() {
            et.background = GradientDrawable().apply {
                cornerRadius = u(10).toFloat(); setColor(Color.argb(255, 16, 18, 28))
                setStroke(max(1, u(2)), if (et.isFocused) Color.WHITE else Color.argb(90, 255, 255, 255))
            }
        }
        stEt()
        et.setOnFocusChangeListener { _, _ -> stEt() }
        val msg = TextView(this).apply { setTextColor(Color.parseColor("#FF5252")); gravity = Gravity.CENTER }
        tx(msg, 15f)
        val btn = TextView(this).apply {
            text = "Verify"; typeface = Typeface.DEFAULT_BOLD; gravity = Gravity.CENTER
            setPadding(u(46), u(13), u(46), u(13))
            isFocusable = true; isFocusableInTouchMode = true; isClickable = true
        }
        tx(btn, 18f)
        fun stBtn() {
            val fo = btn.isFocused
            btn.setTextColor(if (fo) Color.BLACK else Color.WHITE)
            btn.background = GradientDrawable().apply { cornerRadius = u(26).toFloat(); setColor(if (fo) Color.WHITE else Color.argb(160, 90, 94, 120)) }
        }
        stBtn()
        btn.setOnFocusChangeListener { _, _ -> stBtn() }
        val submit: () -> Unit = {
            val tk = et.text.toString().trim()
            if (tk.isEmpty()) { msg.setTextColor(Color.parseColor("#FF5252")); msg.text = "Enter an access token" }
            else if (!gateBusy) {
                gateBusy = true
                msg.setTextColor(Color.parseColor("#B4B8C8")); msg.text = "Checking..."
                Thread {
                    val ok = gateCheck(tk)
                    ui.post {
                        gateBusy = false
                        when (ok) {
                            true -> {
                                getSharedPreferences("liv", Context.MODE_PRIVATE).edit().putString("acc_token", tk).apply()
                                gateOn = false
                                recreate()                      // fresh start, opens the app
                            }
                            false -> { msg.setTextColor(Color.parseColor("#FF5252")); msg.text = "Unauthorized access. Invalid access token" }
                            null -> { msg.setTextColor(Color.parseColor("#FFC107")); msg.text = "Could not verify. Check your internet connection" }
                        }
                    }
                }.start()
            }
        }
        btn.setOnClickListener { submit() }
        et.setOnEditorActionListener { _, _, _ -> submit(); true }
        col.addView(lock, LinearLayout.LayoutParams(-2, -2))
        col.addView(t1, LinearLayout.LayoutParams(-2, -2).apply { topMargin = u(8) })
        col.addView(t2, LinearLayout.LayoutParams(-2, -2).apply { topMargin = u(8) })
        col.addView(et, LinearLayout.LayoutParams(u(440), -2).apply { topMargin = u(22) })
        col.addView(msg, LinearLayout.LayoutParams(-2, -2).apply { topMargin = u(10) })
        col.addView(btn, LinearLayout.LayoutParams(-2, -2).apply { topMargin = u(16) })
        f.addView(col, lp(-2, -2, Gravity.CENTER))
        root.addView(f, lp(-1, -1))
        f.bringToFront()
        gateView = f
        ui.postDelayed({ et.requestFocus() }, 150)
    }

    private fun isHomeTab() = homeTabs.containsKey(tab)'''
