def rep(old, new, must=False):
    global t
    n = t.count(old)
    if n != 1:

            sys.exit('PATCH FAILED (%d): %s' % (n, old[:110]))
        print('WARN (%d): %s' % (n, old[:90]))
        return
    t = t.replace(old, new)

class MobNestRv(c: Context) : RecyclerView(c) {
    private var x0 = 0f
    private var y0 = 0f
    private val slop = ViewConfiguration.get(c).scaledTouchSlop
    override fun dispatchTouchEvent(e: MotionEvent): Boolean {
        when (e.actionMasked) {
            MotionEvent.ACTION_DOWN -> { x0 = e.x; y0 = e.y; parent?.requestDisallowInterceptTouchEvent(true) }
            MotionEvent.ACTION_MOVE -> {
                val dx = abs(e.x - x0)
                val dy = abs(e.y - y0)
                if (dy > slop && dy > dx * 1.7f) parent?.requestDisallowInterceptTouchEvent(false)
                else if (dx > slop) parent?.requestDisallowInterceptTouchEvent(true)
            }
        }
        return super.dispatchTouchEvent(e)
    }
}

    // ---------------- spotlight: looping 1 2 3 4 5 6 1 2 3 ..., always settles centered
    private fun mobHeroStart(hrv: RecyclerView, n: Int) {
        if (n <= 1) return
        val start = 300000 - 300000 % n
        hrv.post { (hrv.layoutManager as? LinearLayoutManager)?.scrollToPositionWithOffset(start, 0) }
    }

    private fun mobHeroSettle(hrv: RecyclerView) {
        hrv.addOnScrollListener(object : RecyclerView.OnScrollListener() {
            override fun onScrollStateChanged(r: RecyclerView, s: Int) {
                if (s != RecyclerView.SCROLL_STATE_IDLE) return
                val w = r.width
                if (w <= 0) return
                var best = Int.MAX_VALUE
                for (i in 0 until r.childCount) {
                    val c = r.getChildAt(i)
                    val d = (c.left + c.right) / 2 - w / 2
                    if (best == Int.MAX_VALUE || abs(d) < abs(best)) best = d
                }
                if (best != Int.MAX_VALUE && abs(best) > dp(2)) r.smoothScrollBy(best, 0)
            }
        })
    }
